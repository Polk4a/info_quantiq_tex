% Titres non numérotés (\section*, sommaire, résumés) : même style. Le filet n'est tracé que si le titre
% n'est pas vide (les titres vides servent à masquer le titre automatique de la liste des figures ou du glossaire).
\newsavebox{\insatitbox}
\newcommand{\insatitrenn}[1]{%
  \sbox{\insatitbox}{\ignorespaces#1\unskip}%
  \ifdim\wd\insatitbox=0pt\else #1\par\nobreak\vspace{0.5ex}{\color{insarouge}\hrule height 0.8pt}\fi}
\titleformat{name=\section,numberless}
  {\fontsize{15}{18}\selectfont\bfseries\color{insarouge}}{}{0pt}{\insatitrenn}
\titleformat{\subsection}
  {\fontsize{12}{14.5}\selectfont\bfseries\color{insarouge}}{\thesubsection}{0.7em}{}
\titleformat{\subsubsection}
  {\fontsize{10.5}{13}\selectfont\bfseries\color{insagris}}{\thesubsubsection}{0.7em}{}
\titleformat{\paragraph}[runin]
  {\normalfont\normalsize\bfseries\color{insagris}}{\theparagraph}{0.7em}{}
\titlespacing*{\paragraph}{0pt}{1.5ex plus .5ex minus .2ex}{0.6em}
\titlespacing*{\section}{0pt}{3ex plus 1ex minus .2ex}{1.8ex plus .2ex}
\titlespacing*{\subsection}{0pt}{2.5ex plus 1ex minus .2ex}{1.2ex plus .2ex}
\titlespacing*{\subsubsection}{0pt}{2ex plus .5ex minus .2ex}{1ex plus .2ex}
