\section{Notation de Dirac et espace de Hilbert}\label{chap:dirac}
% =====================================================================

Le chapitre~\ref{chap:histoire} décrit l'état d'une particule par une fonction d'onde. Ce
chapitre le reformule avec l'algèbre linéaire : un état est un \emph{vecteur} d'un espace de
Hilbert, et la notation de Dirac \parencite{dirac1939} permet de calculer sans écrire les
composantes. On y définit ensuite le \gls{qubit}, système à deux états qui sert de brique de
base à l'informatique quantique, puis la mesure par projection.

% ---------------------------------------------------------------------
\subsection{Notation de Dirac}\label{sec:dirac}
% ---------------------------------------------------------------------

Un état s'écrit comme un vecteur colonne, par exemple
$\begin{pmatrix} a_1\\ a_2\end{pmatrix}\in\mathbb{C}^2$ : c'est un élément d'un
espace de Hilbert de dimension~2, noté $\mathcal{H}^2$. Dirac le note simplement
$\ket{a}$ et évite d'écrire les composantes à chaque ligne.

\begin{definition}{Notation de Dirac : ket, bra et bra-ket}
\begin{center}
\begin{tabular}{@{}r@{\hspace{0.5em}}l@{\hspace{2em}}l@{}}
  {\Large\color{insarouge}$\ket{a}$}
    & $=\begin{pmatrix}a_1\\ a_2\end{pmatrix}\in\mathcal{H}^2$
    & \textbf{ket} : vecteur colonne \\[1.6ex]
  {\Large\color{insarouge}$\bra{a}$}
    & $=\big(\ket{a}\big)^\dagger=\begin{pmatrix}a_1^* & a_2^*\end{pmatrix}$
    & \textbf{bra} : vecteur ligne \\[1.6ex]
  {\Large\color{insarouge}$\braket{a}{b}$}
    & $=\begin{pmatrix}a_1^* & a_2^*\end{pmatrix}\begin{pmatrix}b_1\\ b_2\end{pmatrix}
       =a_1^*b_1+a_2^*b_2$
    & \textbf{bra-ket} : produit scalaire \\
\end{tabular}
\end{center}
Le bra est le transposé \emph{conjugué} du ket. Le nom vient de l'anglais
\emph{bracket} (crochet) : en accolant $\bra{a}$~(\emph{bra}) et $\ket{b}$~(\emph{ket})
on obtient $\braket{a}{b}$~(\emph{bracket}), un nombre complexe.

Propriétés : $\braket{b}{a}=\braket{a}{b}^*$ et
$\braket{a}{a}=|a_1|^2+|a_2|^2\in\mathbb{R}_+$ (norme au carré).
\end{definition}

Dans l'autre ordre, le produit d'un ket par un bra est une matrice (un opérateur) :
\[
  \ket{a}\bra{b}
  =\begin{pmatrix}a_1\\ a_2\end{pmatrix}\begin{pmatrix}b_1^* & b_2^*\end{pmatrix}
  =\begin{pmatrix}a_1b_1^* & a_1b_2^*\\ a_2b_1^* & a_2b_2^*\end{pmatrix}.
\]

\begin{remarque}[Pourquoi le conjugué ?]
Pour $\ket{a}=\dfrac{1}{\sqrt2}\begin{pmatrix}1\\ \ii\end{pmatrix}$, on a
$\bra{a}=\dfrac{1}{\sqrt2}\begin{pmatrix}1 & -\ii\end{pmatrix}$ et
$\braket{a}{a}=\tfrac12\,(1+1)=1$. Sans conjugaison on trouverait
$\tfrac12\,(1+\ii^2)=0$, ce qui n'est pas une norme.
\end{remarque}

Un état physique est \emph{normalisé} :
\[
  \braket{\Psi}{\Psi}=1 .
\]
Pour une particule dans l'espace, $\ket{\Psi}$ est une fonction de $x$ (dimension
infinie) et le produit scalaire devient une intégrale,
$\braket{\varphi}{\psi}=\int\varphi^*\psi\,\dd x$. On retrouve la condition de
normalisation de Born (section~\ref{sec:born}) :
\[
  \braket{\Psi}{\Psi} = \int_{-\infty}^{+\infty}\Psi^*\Psi\,\dd x = 1 .
\]
La section~\ref{sec:hilbert} précise ce qu'est un espace de Hilbert.

% ---------------------------------------------------------------------
\subsection{Espace de Hilbert}\label{sec:hilbert}
% ---------------------------------------------------------------------

Les kets sont des éléments d'un \glslink{hilbert}{espace de Hilbert} $\mathcal{H}$, cadre
mathématique de la mécanique quantique.

\begin{definition}{Espace de Hilbert}
C'est un espace vectoriel sur $\mathbb{C}$ muni d'un produit scalaire $\braket{a}{b}\in\mathbb{C}$
qui vérifie
\begin{itemize}[nosep]
  \item $\braket{b}{a}=\braket{a}{b}^*$ ;
  \item la linéarité à droite : $\braket{a}{\lambda b+\mu c}=\lambda\braket{a}{b}+\mu\braket{a}{c}$
        (donc l'antilinéarité à gauche : un scalaire sorti d'un bra est conjugué) ;
  \item $\braket{a}{a}\geq0$, avec égalité seulement si $\ket{a}=0$.
\end{itemize}
La \emph{norme} est $\|a\|=\sqrt{\braket{a}{a}}$. L'espace est \emph{complet} : toute suite de
Cauchy y converge (c'est automatique en dimension finie). Deux vecteurs sont \emph{orthogonaux}
si $\braket{a}{b}=0$, et un vecteur est \emph{unitaire} si $\braket{a}{a}=1$.
\end{definition}

\begin{resultat}{Postulat 1 : l'état d'un système}
L'état d'un système quantique isolé est un vecteur unitaire $\ket{\Psi}$ d'un espace de Hilbert
$\mathcal{H}$ : $\braket{\Psi}{\Psi}=1$. Les vecteurs $\ket{\Psi}$ et $\ee^{\ii\varphi}\ket{\Psi}$
décrivent le même état physique : la phase globale n'est pas observable.
\end{resultat}

\paragraph{Base orthonormée.} Une famille $\{\ket{e_i}\}$ est une base orthonormée si
$\braket{e_i}{e_j}=\delta_{ij}$ et si tout vecteur s'en déduit par combinaison linéaire. Pour
$\ket{a}=\sum_j c_j\ket{e_j}$, multiplier par $\bra{e_i}$ isole le coefficient :
\[
  \braket{e_i}{a}=\sum_j c_j\,\braket{e_i}{e_j}=\sum_j c_j\,\delta_{ij}=c_i .
\]

\begin{resultat}{Décomposition sur une base orthonormée}
\[
  \ket{a}=\sum_i c_i\ket{e_i},\qquad c_i=\braket{e_i}{a},\qquad
  \braket{a}{b}=\sum_i a_i^*\,b_i,\qquad \braket{a}{a}=\sum_i|c_i|^2 .
\]
Comme $\ket{a}=\sum_i\ket{e_i}\braket{e_i}{a}$ pour tout $\ket{a}$, on obtient la
\emph{relation de fermeture}
\[
  \sum_i\ket{e_i}\bra{e_i}=\mathrm{Id}.
\]
\end{resultat}

Le vecteur $\ket{a}$ est un objet géométrique : ses composantes $c_i=\braket{e_i}{a}$ sont ses
projections sur les axes de la base, et elles changent quand on change de base. La
figure~\ref{fig:vecteur} l'illustre avec des amplitudes réelles, pour pouvoir dessiner le vecteur
dans le plan.

\paragraph{Dimension finie et dimension infinie.} Le même formalisme décrit un qubit
(dimension~2) et une particule dans un puits (dimension infinie). Le tableau les compare.

\begin{center}
\renewcommand{\arraystretch}{1.55}
\begin{tabular}{@{}lll@{}}
\toprule
 & \textbf{Dimension finie : $\mathbb{C}^2$} & \textbf{Dimension infinie : $L^2([0,a])$} \\
\midrule
Vecteur & $\ket{a}=(a_1,a_2)^{\mathsf{T}}$ & $\ket{\Psi}\leftrightarrow\Psi(x)$ \\
Produit scalaire & $\braket{a}{b}=\sum_i a_i^*\,b_i$
  & $\braket{\varphi}{\psi}=\displaystyle\int_0^a\varphi^*\psi\,\dd x$ \\
Base orthonormée & $\ket{0},\ \ket{1}$ & $\ket{n}\leftrightarrow\psi_n(x)=\sqrt{\dfrac2a}\sin\dfrac{n\pi x}{a}$ \\
Coefficients & $c_i=\braket{i}{a}$ & $c_n=\braket{n}{\Psi}=\displaystyle\int_0^a\psi_n^*\,\Psi\,\dd x$ \\
Exemple & qubit & particule dans le puits infini \\
\bottomrule
\end{tabular}
\end{center}

L'espace $L^2$ est celui des fonctions de carré sommable, $\int|\Psi|^2\dd x<\infty$ : la
condition de normalisation de Born s'écrit $\braket{\Psi}{\Psi}=1$. Les états stationnaires du
puits infini forment une base orthonormée (section~\ref{sec:solutions}), et la superposition
$\Psi=\sum_n c_n\psi_n$ de la section~\ref{sec:superposition} est la décomposition de $\ket{\Psi}$
sur cette base, avec $P(E_n)=|c_n|^2$.

\begin{remarque}[Cauchy--Schwarz et probabilités]
Pour deux vecteurs quelconques, $|\braket{a}{b}|^2\leq\braket{a}{a}\,\braket{b}{b}$. Pour deux
vecteurs unitaires, $|\braket{a}{b}|^2$ est donc un nombre entre $0$ et $1$, égal à $1$ seulement si
$\ket{a}$ et $\ket{b}$ sont proportionnels : c'est la probabilité de trouver le système dans
l'état $\ket{b}$ s'il est dans l'état $\ket{a}$ (section~\ref{sec:projecteurs}).
\end{remarque}

% ---------------------------------------------------------------------
\subsection{Le qubit}\label{sec:qubit}
% ---------------------------------------------------------------------

Reprenons l'exemple de la section~\ref{sec:superposition} en ne gardant que deux états du puits,
notés comme un bit quantique (\gls{qubit}) :
$\ket{0}$ pour $\Psi_1$ et $\ket{1}$ pour $\Psi_2$. L'état de l'exemple s'écrit
\[
  \ket{\Psi}=\frac{1}{\sqrt2}\,\ket{0}+\frac{1}{\sqrt2}\,\ket{1}.
\]

\begin{center}
\renewcommand{\arraystretch}{1.85}
\begin{tabular}{@{}ll@{}}
\toprule
\textbf{Fonctions d'onde} & \textbf{Notation de Dirac} \\
\midrule
$\Psi=c_1\Psi_1+c_2\Psi_2$ & $\ket{\Psi}=c_1\ket{0}+c_2\ket{1}$ \\
$\displaystyle\int\Psi_i^*\Psi_j\,\dd x=\delta_{ij}$ & $\braket{i}{j}=\delta_{ij}$ \\
$\displaystyle\int\Psi^*\Psi\,\dd x=\sum_i|c_i|^2=1$ & $\braket{\Psi}{\Psi}=1$ \\
$P(E_i)=|c_i|^2$ & $P(i)=|c_i|^2=|\braket{i}{\Psi}|^2$ \\
\bottomrule
\end{tabular}
\end{center}

\paragraph{Probabilité.} La probabilité que $\ket{\Psi}$ soit mesuré dans l'état
$\ket{0}$ est
\[
  P(\ket{\Psi}\to\ket{0})=P(0)=\left|\frac{1}{\sqrt2}\right|^2=\frac12 .
\]

\paragraph{Normalisation.}
\[
  \braket{\Psi}{\Psi}
  =\Big(\tfrac{1}{\sqrt2}\bra{0}+\tfrac{1}{\sqrt2}\bra{1}\Big)
   \Big(\tfrac{1}{\sqrt2}\ket{0}+\tfrac{1}{\sqrt2}\ket{1}\Big)
  =\tfrac12\braket{0}{0}+\tfrac12\braket{0}{1}+\tfrac12\braket{1}{0}+\tfrac12\braket{1}{1}
  =\tfrac12\,(1+0+0+1)=1 .
\]

\begin{definition}{Base de calcul (base $Z$)}
$\{\ket{0},\ket{1}\}$ est une base \emph{orthonormée} de $\mathcal{H}^2=\mathbb{C}^2$ :
\[
  \ket{0}=\begin{pmatrix}1\\ 0\end{pmatrix},\qquad
  \ket{1}=\begin{pmatrix}0\\ 1\end{pmatrix},\qquad
  \braket{0}{0}=\begin{pmatrix}1 & 0\end{pmatrix}\begin{pmatrix}1\\ 0\end{pmatrix}=1,\qquad
  \braket{0}{1}=\begin{pmatrix}1 & 0\end{pmatrix}\begin{pmatrix}0\\ 1\end{pmatrix}=0 .
\]
Tout état s'écrit $\ket{\Psi}=\alpha\ket{0}+\beta\ket{1}=\begin{pmatrix}\alpha\\ \beta\end{pmatrix}$
avec $|\alpha|^2+|\beta|^2=1$ ; alors $P(0)=|\alpha|^2$ et $P(1)=|\beta|^2$.
\end{definition}

\paragraph{Exemple.} Soit $\ket{\Psi}=\dfrac{1}{\sqrt5}\,\ket{0}+\dfrac{2}{\sqrt5}\,\ket{1}$
dans $\mathcal{H}^2$.
\begin{enumerate}[nosep]
  \item \emph{$\ket{\Psi}$ est-il normalisé ?}
        $\braket{\Psi}{\Psi}=\dfrac15+\dfrac45=1$ : oui.
  \item $P(0)=\left|\dfrac{1}{\sqrt5}\right|^2=\dfrac15$.
  \item $P(1)=\left|\dfrac{2}{\sqrt5}\right|^2=\dfrac45$.
\end{enumerate}

\begin{remarque}[État non normalisé]
Si $\braket{\Psi}{\Psi}\neq1$, on divise l'état par sa norme $\sqrt{\braket{\Psi}{\Psi}}$.
Par exemple $\ket{0}+2\ket{1}$ a pour norme au carré $1+4=5$ : l'état normalisé est
$\frac{1}{\sqrt5}\ket{0}+\frac{2}{\sqrt5}\ket{1}$, celui de l'exemple.
\end{remarque}

\paragraph{Autres bases orthonormées.} La base de calcul n'est pas la seule base de $\mathbb{C}^2$.
Les états
\[
  \ket{\pm}=\frac{1}{\sqrt2}\big(\ket{0}\pm\ket{1}\big)\quad(\text{base }X),\qquad
  \ket{{\pm}\ii}=\frac{1}{\sqrt2}\big(\ket{0}\pm\ii\ket{1}\big)\quad(\text{base }Y)
\]
forment deux autres bases orthonormées. Par exemple $\braket{+}{-}=\frac12(1-1)=0$ et, en
conjuguant le bra, $\braket{{+}\ii}{{-}\ii}=\frac12\big(1+(-\ii)(-\ii)\big)=\frac12(1-1)=0$.
Réciproquement $\ket{0}=\frac{1}{\sqrt2}(\ket{+}+\ket{-})$ et
$\ket{1}=\frac{1}{\sqrt2}(\ket{+}-\ket{-})$ : un état de la base $Z$ est une superposition d'états
de la base $X$, et inversement. La sphère de Bloch (section~\ref{sec:bloch}) place ces six états sur
trois axes.

% ---------------------------------------------------------------------
\subsection{Projecteurs et mesure}\label{sec:projecteurs}
% ---------------------------------------------------------------------

Le produit d'un ket par un bra est un opérateur (section~\ref{sec:dirac}). Avec les
états de la base de calcul, on obtient deux \emph{projecteurs}.

\begin{definition}{Projecteurs sur $\ket{0}$ et sur $\ket{1}$}
\[
  \hat P_0=\ket{0}\bra{0}=\begin{pmatrix}1\\ 0\end{pmatrix}\begin{pmatrix}1 & 0\end{pmatrix}
  =\begin{pmatrix}1 & 0\\ 0 & 0\end{pmatrix},
  \qquad
  \hat P_1=\ket{1}\bra{1}=\begin{pmatrix}0\\ 1\end{pmatrix}\begin{pmatrix}0 & 1\end{pmatrix}
  =\begin{pmatrix}0 & 0\\ 0 & 1\end{pmatrix}.
\]
Ils vérifient
\[
  \hat P_i^{\,2}=\hat P_i,
  \qquad
  \hat P_i^{\,\dagger}=\hat P_i,
  \qquad
  \hat P_0+\hat P_1=\begin{pmatrix}1 & 0\\ 0 & 1\end{pmatrix}=\mathrm{Id}
  \quad(\text{complétude}).
\]
\end{definition}

En effet $\hat P_0^{\,2}=\ket{0}\braket{0}{0}\bra{0}=\ket{0}\bra{0}$ car
$\braket{0}{0}=1$, et $\hat P_0^{\,\dagger}=(\ket{0}\bra{0})^\dagger=\ket{0}\bra{0}$ : un opérateur
hermitien qui égale son carré est un projecteur. Appliqué à $\ket{\Psi}=\alpha\ket{0}+\beta\ket{1}$, $\hat P_0$
ne garde que la composante sur $\ket{0}$ : $\hat P_0\ket{\Psi}=\alpha\ket{0}$.

\begin{resultat}{Postulat 2 : probabilité de mesure (règle de Born)}
\[
  P(\ket{\Psi}\to\ket{i})=\bra{\Psi}\hat P_i\ket{\Psi}
  =\braket{\Psi}{i}\braket{i}{\Psi}=|\braket{i}{\Psi}|^2 .
\]
La complétude donne bien $\sum_i P(i)=\bra{\Psi}\big(\sum_i\hat P_i\big)\ket{\Psi}=\braket{\Psi}{\Psi}=1$.
Après la mesure, l'état du système est $\ket{i}$ (\emph{effondrement}, figure~\ref{fig:mesure}).
\end{resultat}

C'est la règle de Born \parencite{born1926} de la section~\ref{sec:born}, écrite pour un système
à états discrets.

\paragraph{Exemple.} Pour $\ket{\Psi}=\dfrac{1}{\sqrt5}\ket{0}+\dfrac{2}{\sqrt5}\ket{1}
=\begin{pmatrix}1/\sqrt5\\ 2/\sqrt5\end{pmatrix}$ :
\[
  \hat P_0\ket{\Psi}=\begin{pmatrix}1 & 0\\ 0 & 0\end{pmatrix}
  \begin{pmatrix}1/\sqrt5\\ 2/\sqrt5\end{pmatrix}
  =\begin{pmatrix}1/\sqrt5\\ 0\end{pmatrix},
  \qquad
  P(\ket{\Psi}\to\ket{0})=\bra{\Psi}\hat P_0\ket{\Psi}
  =\begin{pmatrix}\frac{1}{\sqrt5} & \frac{2}{\sqrt5}\end{pmatrix}
   \begin{pmatrix}\frac{1}{\sqrt5}\\ 0\end{pmatrix}=\frac15 ,
\]
\[
  P(\ket{\Psi}\to\ket{1})=\bra{\Psi}\hat P_1\ket{\Psi}
  =\begin{pmatrix}\frac{1}{\sqrt5} & \frac{2}{\sqrt5}\end{pmatrix}
   \begin{pmatrix}0 & 0\\ 0 & 1\end{pmatrix}
   \begin{pmatrix}\frac{1}{\sqrt5}\\ \frac{2}{\sqrt5}\end{pmatrix}=\frac45 .
\]

\end{figure}

\paragraph{Mesure dans une autre base.} La règle de Born vaut pour toute base orthonormée. Pour la base
$X$, les projecteurs sont $\hat P_\pm=\ket{\pm}\bra{\pm}$ et $P(\pm)=|\braket{\pm}{\Psi}|^2$. Pour
$\ket{\Psi}=\frac{1}{\sqrt5}\ket{0}+\frac{2}{\sqrt5}\ket{1}$,
\[
  \braket{+}{\Psi}=\frac{1}{\sqrt2}\Big(\frac{1}{\sqrt5}+\frac{2}{\sqrt5}\Big)=\frac{3}{\sqrt{10}},\qquad
  \braket{-}{\Psi}=\frac{1}{\sqrt2}\Big(\frac{1}{\sqrt5}-\frac{2}{\sqrt5}\Big)=-\frac{1}{\sqrt{10}},
\]
donc $P(+)=\frac{9}{10}$ et $P(-)=\frac{1}{10}$ (figure~\ref{fig:vecteur}b), de somme $1$. Le même état
donne des probabilités différentes selon la base dans laquelle on le mesure.

\paragraph{Effet d'une phase relative.} Soit $\ket{\Psi'}=\dfrac{1}{\sqrt5}\ket{0}
+\dfrac{2\ii}{\sqrt5}\ket{1}$. Son bra est le \emph{conjugué} :
$\bra{\Psi'}=\begin{pmatrix}\frac{1}{\sqrt5} & -\frac{2\ii}{\sqrt5}\end{pmatrix}$, donc
\[
  \braket{\Psi'}{\Psi'}=\frac15+\Big(-\frac{2\ii}{\sqrt5}\Big)\Big(\frac{2\ii}{\sqrt5}\Big)
  =\frac15+\frac45=1 ,
\]
et $P(0)=1/5$, $P(1)=4/5$ comme pour $\ket{\Psi}$. Les deux états sont pourtant
différents : le facteur $\ii$ est une phase \emph{relative} entre les deux composantes,
que la mesure dans la base $Z$ ne voit pas (contrairement à une phase globale, jamais
observable). La base $X$ la révèle : $\braket{\pm}{\Psi'}=(1\pm2\ii)/\sqrt{10}$, donc
$P(+)=P(-)=\frac12$, alors que $\ket{\Psi}$ donne $(\frac{9}{10},\frac{1}{10})$
(figure~\ref{fig:histo}).

\begin{remarque}[Discerner deux états : seuls des états orthogonaux le sont à coup sûr]\label{rem:discerner}
Deux états \emph{orthogonaux} ($\braket{a}{b}=0$) se distinguent parfaitement : on mesure dans une
base qui les contient et le résultat dit lequel c'était. Si $\braket{a}{b}\neq0$, aucune mesure ne
suffit. Pour $\ket{0}$ et $\ket{+}$ ($\braket{0}{+}=\frac{1}{\sqrt2}$), une mesure en base $Z$ donne
$0$ avec la probabilité $1$ pour $\ket{0}$ mais aussi avec la probabilité $\frac12$ pour $\ket{+}$ :
le résultat $0$ ne tranche pas (le résultat $1$ prouve seulement qu'on avait $\ket{+}$). La mesure a
de plus effondré l'état : on ne peut pas recommencer sur le même qubit.
\end{remarque}

\begin{aretenir}
\begin{itemize}
  \item Un ket est un vecteur colonne, le bra est son transposé \emph{conjugué}. Le bra-ket
        $\braket{a}{b}$ est un nombre ; le ket-bra $\ket{a}\bra{b}$ est une matrice.
  \item Un espace de Hilbert est un espace vectoriel complexe muni d'un produit scalaire, complet.
        Sur une base orthonormée, $c_i=\braket{e_i}{a}$ et $\sum_i\ket{e_i}\bra{e_i}=\mathrm{Id}$.
  \item Un état est un vecteur unitaire, $\braket{\Psi}{\Psi}=1$, défini à une phase globale près.
        Une fonction d'onde est un ket de dimension infinie ($L^2$).
  \item Un qubit est un état $\alpha\ket{0}+\beta\ket{1}$ de $\mathbb{C}^2$ avec $|\alpha|^2+|\beta|^2=1$ ;
        les bases $Z$, $X$ et $Y$ sont trois bases orthonormées de $\mathbb{C}^2$.
  \item Mesurer dans une base $\{\ket{i}\}$ donne $i$ avec la probabilité
        $P(i)=\bra{\Psi}\hat P_i\ket{\Psi}=|\braket{i}{\Psi}|^2$, puis l'état devient $\ket{i}$.
  \item La phase globale n'est jamais observable. La phase relative est invisible en base $Z$ mais
        pas dans une autre base (figure~\ref{fig:histo}).
\end{itemize}
\end{aretenir}
