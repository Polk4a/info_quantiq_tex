\section{Opérateurs, matrice de densité, sphère de Bloch et produit tensoriel}\label{chap:operateurs}
% =====================================================================

Le chapitre~\ref{chap:dirac} décrit l'état d'un qubit. Ce chapitre montre ce que l'on peut faire
avec cet état (les \emph{opérateurs} : mesure d'une grandeur, évolution), comment décrire ce que
l'on ignore de lui (la \emph{matrice de densité}, les états mixtes et la sphère de Bloch), puis
comment réunir plusieurs qubits (le \emph{produit tensoriel}).

% ---------------------------------------------------------------------
\subsection{Opérateurs quantiques}\label{sec:op-dirac}
% ---------------------------------------------------------------------

Au chapitre~\ref{chap:histoire}, un opérateur transformait une fonction d'onde
(section~\ref{sec:operateurs}). Pour un qubit, c'est une matrice $2\times2$. Les projecteurs
$\hat P_0=\ket{0}\bra{0}$ et $\hat P_1=\ket{1}\bra{1}$ (section~\ref{sec:projecteurs}) sont des
cas particuliers : tout opérateur $A$ s'écrit comme une somme de ket-bra.

\begin{definition}{Décomposition d'un opérateur sur la base $\{\ket{0},\ket{1}\}$}
\[
  A=\begin{pmatrix}A_{00} & A_{01}\\ A_{10} & A_{11}\end{pmatrix}
  =A_{00}\ket{0}\bra{0}+A_{01}\ket{0}\bra{1}+A_{10}\ket{1}\bra{0}+A_{11}\ket{1}\bra{1},
  \qquad A_{ij}=\bra{i}A\ket{j}.
\]
\end{definition}

Chaque ket-bra est une matrice qui ne contient qu'un seul $1$, à la ligne $i$ et à la
colonne $j$ :
\[
  \ket{0}\bra{1}=\begin{pmatrix}1\\ 0\end{pmatrix}\begin{pmatrix}0 & 1\end{pmatrix}
  =\begin{pmatrix}0 & 1\\ 0 & 0\end{pmatrix},
  \qquad
  \ket{1}\bra{0}=\begin{pmatrix}0\\ 1\end{pmatrix}\begin{pmatrix}1 & 0\end{pmatrix}
  =\begin{pmatrix}0 & 0\\ 1 & 0\end{pmatrix}.
\]

\paragraph{Exemples.} Les trois matrices de Pauli s'écrivent
\begin{align*}
  X&=\ket{0}\bra{1}+\ket{1}\bra{0}=\begin{pmatrix}0 & 1\\ 1 & 0\end{pmatrix},\\
  Y&=-\ii\ket{0}\bra{1}+\ii\ket{1}\bra{0}=\begin{pmatrix}0 & -\ii\\ \ii & 0\end{pmatrix},\\
  Z&=\ket{0}\bra{0}-\ket{1}\bra{1}=\begin{pmatrix}1 & 0\\ 0 & -1\end{pmatrix}.
\end{align*}
Leur action se lit directement sur l'écriture ket-bra, grâce à
$\braket{0}{0}=\braket{1}{1}=1$ et $\braket{0}{1}=0$ :
$X\ket{0}=\ket{0}\braket{1}{0}+\ket{1}\braket{0}{0}=\ket{1}$ (et de même $X\ket{1}=\ket{0}$) :
$X$ échange $\ket{0}$ et $\ket{1}$ ; $Z\ket{0}=\ket{0}$ et $Z\ket{1}=-\ket{1}$.

\paragraph{Adjoint, opérateurs hermitiens et unitaires.} L'\emph{adjoint} $A^\dagger$ d'un opérateur
est sa transposée conjuguée : $(A^\dagger)_{ij}=A_{ji}^*$, soit $\bra{a}A^\dagger\ket{b}=\bra{b}A\ket{a}^*$.
On a $(\ket{a}\bra{b})^\dagger=\ket{b}\bra{a}$ et $(AB)^\dagger=B^\dagger A^\dagger$.

\begin{definition}{Opérateurs hermitiens et unitaires}
\begin{itemize}[nosep]
  \item $A$ est \gls{hermitien} si $A^\dagger=A$. Ses valeurs propres sont réelles : si
        $A\ket{a}=\lambda\ket{a}$, alors $\lambda\braket{a}{a}=\bra{a}A\ket{a}$ est réel, car
        $\bra{a}A\ket{a}^*=\bra{a}A^\dagger\ket{a}=\bra{a}A\ket{a}$. Deux vecteurs propres de valeurs
        propres différentes sont orthogonaux.
  \item $U$ est \gls{unitaire} si $U^\dagger U=UU^\dagger=\mathrm{Id}$. Il conserve la norme :
        $\big(U\ket{\Psi}\big)^\dagger U\ket{\Psi}=\bra{\Psi}U^\dagger U\ket{\Psi}=\braket{\Psi}{\Psi}$.
\end{itemize}
\end{definition}

Les matrices de Pauli sont à la fois hermitiennes et unitaires : $X^\dagger=X$, $Y^\dagger=Y$,
$Z^\dagger=Z$ et $X^2=Y^2=Z^2=\mathrm{Id}$, donc $X^\dagger X=\mathrm{Id}$. Leurs valeurs propres sont
$+1$ et $-1$, et leurs vecteurs propres sont les états des trois bases de la
section~\ref{sec:qubit}.

\begin{center}
\renewcommand{\arraystretch}{1.5}
\begin{tabular}{@{}c c c c@{}}
\toprule
\textbf{Observable} & \textbf{Écriture spectrale} & \textbf{Vecteur propre $+1$}
  & \textbf{Vecteur propre $-1$} \\
\midrule
$Z$ & $\ket{0}\bra{0}-\ket{1}\bra{1}$ & $\ket{0}$ & $\ket{1}$ \\
$X$ & $\ket{+}\bra{+}-\ket{-}\bra{-}$ & $\ket{+}$ & $\ket{-}$ \\
$Y$ & $\ket{{+}\ii}\bra{{+}\ii}-\ket{{-}\ii}\bra{{-}\ii}$ & $\ket{{+}\ii}$ & $\ket{{-}\ii}$ \\
\bottomrule
\end{tabular}
\end{center}

Vérification pour $Y$ : $Y\ket{0}=\ii\ket{1}$ et $Y\ket{1}=-\ii\ket{0}$, donc
\[
  Y\ket{{+}\ii}=\frac{1}{\sqrt2}\Big(\ii\ket{1}+\ii\,(-\ii)\ket{0}\Big)
  =\frac{1}{\sqrt2}\big(\ket{0}+\ii\ket{1}\big)=\ket{{+}\ii}.
\]

\begin{resultat}{Observable, valeurs propres et valeur moyenne}
Une grandeur physique est représentée par un opérateur hermitien $A$. Les résultats possibles d'une
mesure sont ses valeurs propres. Si $A=\sum_a a\,\hat P_a$ où $\hat P_a$ est le projecteur sur le
sous-espace propre de valeur propre $a$, alors
\[
  P(a)=\bra{\Psi}\hat P_a\ket{\Psi},\qquad
  \langle A\rangle=\sum_a a\,P(a)=\bra{\Psi}A\ket{\Psi}.
\]
\end{resultat}

C'est la formule $\langle\hat O\rangle=\int\Psi^*\hat O\,\Psi\,\dd x$ de la
section~\ref{sec:operateurs}, écrite avec la notation de Dirac. Le tableau résume le lien entre
les deux chapitres.

\begin{center}
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{@{}l l l@{}}
\toprule
 & \textbf{Fonctions d'onde (chapitre~\ref{chap:histoire})} & \textbf{Qubit} \\
\midrule
État & $\Psi(x,t)$ & vecteur $\ket{\Psi}\in\mathbb{C}^2$ \\
Grandeur & opérateur $\hat O$ ($\hat p$, $\hat H$) & matrice hermitienne $A$ \\
Résultats possibles & valeurs propres de $\hat O$ & valeurs propres de $A$ \\
Valeur moyenne & $\int\Psi^*\hat O\,\Psi\,\dd x$ & $\bra{\Psi}A\ket{\Psi}$ \\
\bottomrule
\end{tabular}
\end{center}

\paragraph{Exemple.} Pour $\ket{\Psi}=\frac{1}{\sqrt5}\ket{0}+\frac{2}{\sqrt5}\ket{1}$,
\begin{align*}
  \langle Z\rangle&=\bra{\Psi}Z\ket{\Psi}
    =\begin{pmatrix}\frac{1}{\sqrt5} & \frac{2}{\sqrt5}\end{pmatrix}
     \begin{pmatrix}\frac{1}{\sqrt5}\\ -\frac{2}{\sqrt5}\end{pmatrix}=\frac15-\frac45=-\frac35,\\[1ex]
  \langle X\rangle&=\begin{pmatrix}\frac{1}{\sqrt5} & \frac{2}{\sqrt5}\end{pmatrix}
     \begin{pmatrix}\frac{2}{\sqrt5}\\ \frac{1}{\sqrt5}\end{pmatrix}=\frac25+\frac25=\frac45,\\[1ex]
  \langle Y\rangle&=\begin{pmatrix}\frac{1}{\sqrt5} & \frac{2}{\sqrt5}\end{pmatrix}
     \begin{pmatrix}-\frac{2\ii}{\sqrt5}\\ \frac{\ii}{\sqrt5}\end{pmatrix}=\frac{-2\ii+2\ii}{5}=0 .
\end{align*}
Ces trois nombres seront les coordonnées de l'état sur la sphère de Bloch (section~\ref{sec:bloch}).
On retrouve $\langle Z\rangle=P(0)-P(1)$ et $\langle X\rangle=P(+)-P(-)=\frac{9}{10}-\frac{1}{10}$.

% ---------------------------------------------------------------------
\subsection{Matrice de densité}\label{sec:rho}
% ---------------------------------------------------------------------

\begin{definition}{Matrice de densité d'un état pur}
Pour $\ket{\Psi}=\alpha\ket{0}+\beta\ket{1}$ avec $|\alpha|^2+|\beta|^2=1$ :
\[
  \rho=\ket{\Psi}\bra{\Psi}
  =\begin{pmatrix}\alpha\\ \beta\end{pmatrix}\begin{pmatrix}\alpha^* & \beta^*\end{pmatrix}
  =\begin{pmatrix}|\alpha|^2 & \alpha\beta^*\\ \alpha^*\beta & |\beta|^2\end{pmatrix}.
\]
\end{definition}

La diagonale contient les probabilités de mesure ($\rho_{00}=|\alpha|^2=P(0)$,
$\rho_{11}=|\beta|^2=P(1)$, ce sont les \emph{populations}) ; les termes hors diagonale
(les \emph{cohérences}) contiennent la phase relative entre $\alpha$ et $\beta$.
Une phase globale $\ket{\Psi}\to\ee^{\ii\gamma}\ket{\Psi}$ disparaît de $\rho$ :
$\ee^{\ii\gamma}\ee^{-\ii\gamma}=1$.

\begin{resultat}{Propriétés d'une matrice de densité $\rho$}
Notons $\lambda_i$ les valeurs propres de $\rho$.
\begin{enumerate}[nosep]
  \item $\operatorname{Tr}\rho=1=\displaystyle\sum_i\lambda_i$ ;
  \item $\rho$ est \emph{hermitienne} : $\rho^\dagger=\rho$ (transposée conjuguée) ;
  \item $\rho$ est \emph{positive} : $\bra{\Phi}\rho\ket{\Phi}\geq0$ pour tout
        $\ket{\Phi}$, ce qui équivaut à $\lambda_i\geq0$.
\end{enumerate}
\end{resultat}

Vérification pour $\rho=\ket{\Psi}\bra{\Psi}$ :
$\operatorname{Tr}\rho=|\alpha|^2+|\beta|^2=1$ ; $(\ket{\Psi}\bra{\Psi})^\dagger
=\ket{\Psi}\bra{\Psi}$ ; et $\bra{\Phi}\rho\ket{\Phi}=\braket{\Phi}{\Psi}\braket{\Psi}{\Phi}
=|\braket{\Psi}{\Phi}|^2\geq0$.

\begin{remarque}[Hermitienne, pas seulement symétrique]
Il faut $\rho^\dagger=\rho$ et non $\rho^{\mathsf T}=\rho$ : la seconde égalité n'est vraie
que si $\rho$ est réelle. Ainsi $\ket{\Psi'}\bra{\Psi'}$ avec un facteur $\ii$ (exemple
ci-dessous) est hermitienne mais pas symétrique.
\end{remarque}

\paragraph{Exemples.} Pour $\ket{\Psi}=\frac{1}{\sqrt5}\ket{0}+\frac{2}{\sqrt5}\ket{1}$ et
pour $\ket{\Psi'}=\frac{1}{\sqrt5}\ket{0}+\frac{2\ii}{\sqrt5}\ket{1}$
(section~\ref{sec:projecteurs}) :
\[
  \rho=\begin{pmatrix}\dfrac15 & \dfrac25\\[0.6ex] \dfrac25 & \dfrac45\end{pmatrix},
  \qquad
  \rho'=\begin{pmatrix}\dfrac15 & -\dfrac{2\ii}{5}\\[0.6ex] \dfrac{2\ii}{5} & \dfrac45\end{pmatrix}.
\]
Les diagonales sont identiques (mêmes probabilités $\frac15$ et $\frac45$), seules les
cohérences diffèrent : la matrice de densité voit la phase relative que la mesure en
base $Z$ ne voit pas.

\begin{remarque}[Probabilités et valeurs moyennes avec $\rho$]
$P(i)=\bra{i}\rho\ket{i}=\rho_{ii}$ et $\langle A\rangle=\bra{\Psi}A\ket{\Psi}
=\operatorname{Tr}(\rho A)$. Ces deux formules restent valables pour les états mixtes
de la section~\ref{sec:pur-mixte}.
\end{remarque}

% ---------------------------------------------------------------------
\subsection{État pur ou état mixte}\label{sec:pur-mixte}
% ---------------------------------------------------------------------

Comme $\sum_i\lambda_i=1$ et $0\leq\lambda_i\leq1$, on a
$\operatorname{Tr}\rho^2=\sum_i\lambda_i^2\leq1$, avec égalité si et seulement si une
seule valeur propre vaut $1$ (les autres sont nulles).

\begin{definition}{Pur ou mixte}
\begin{itemize}[nosep]
  \item $\rho$ est \emph{pur} si $\operatorname{Tr}\rho^2=1$ : il existe $a$ avec
        $\lambda_a=1$ et $\lambda_b=0$ pour $b\neq a$. Alors $\rho=\ket{\Psi}\bra{\Psi}$
        et $\rho^2=\rho$.
  \item $\rho$ est \emph{mixte} si $\operatorname{Tr}\rho^2<1$ : c'est un mélange
        statistique $\rho=\sum_ip_i\ket{\Psi_i}\bra{\Psi_i}$ d'états purs, chacun avec la
        probabilité $p_i$ ($\sum_ip_i=1$). On ne sait pas dans lequel des $\ket{\Psi_i}$
        se trouve le système.
\end{itemize}
\end{definition}

\begin{center}
\begin{tabular}{@{}lccl@{}}
\toprule
\textbf{Matrice de densité} & $\operatorname{Tr}\rho$ & $\operatorname{Tr}\rho^2$ & \textbf{Nature} \\
\midrule
$\ket{0}\bra{0}=\begin{pmatrix}1 & 0\\ 0 & 0\end{pmatrix}$ & $1$ & $1$ & pur \\[1.6ex]
$\dfrac12\begin{pmatrix}1 & -1\\ -1 & 1\end{pmatrix}=\ket{-}\bra{-}$ & $1$ & $1$ & pur \\[1.6ex]
$\dfrac12\begin{pmatrix}1 & 0\\ 0 & 1\end{pmatrix}=\dfrac{\mathrm{Id}}{2}$ & $1$ & $\dfrac12$ & mixte \\[1.6ex]
$\begin{pmatrix}\frac34 & 0\\ 0 & \frac14\end{pmatrix}$ & $1$ & $\dfrac58$ & mixte \\
\bottomrule
\end{tabular}
\end{center}

Détails. Pour la deuxième ligne :
$\dfrac12\begin{pmatrix}1 & -1\\ -1 & 1\end{pmatrix}
=\dfrac12\big[\ket{0}\bra{0}-\ket{0}\bra{1}-\ket{1}\bra{0}+\ket{1}\bra{1}\big]$,
qui est bien $\ket{-}\bra{-}$ avec $\ket{-}=\frac{1}{\sqrt2}(\ket{0}-\ket{1})$ ; comme
$\rho^2=\frac14\begin{pmatrix}2 & -2\\ -2 & 2\end{pmatrix}=\rho$, on a
$\operatorname{Tr}\rho^2=1$. Pour la troisième : $\rho^2=\frac14\mathrm{Id}$, donc
$\operatorname{Tr}\rho^2=\frac12$. Pour la quatrième :
$\operatorname{Tr}\rho^2=\frac{9}{16}+\frac1{16}=\frac58$.

\begin{remarque}[Mélange $\neq$ superposition]
$\rho_{\text{mél}}=\frac12\ket{0}\bra{0}+\frac12\ket{1}\bra{1}=\mathrm{Id}/2$ (le système
est $\ket{0}$ ou $\ket{1}$, une fois sur deux) n'est pas l'état pur
$\ket{+}\bra{+}=\frac12\begin{pmatrix}1 & 1\\ 1 & 1\end{pmatrix}$ (superposition
cohérente). Les deux donnent $P(0)=P(1)=\frac12$, mais une mesure dans la base
$\{\ket{+},\ket{-}\}$ les distingue : $\bra{+}\rho_{\text{mél}}\ket{+}=\frac12$, alors que
$\bra{+}\big(\ket{+}\bra{+}\big)\ket{+}=1$. Ce sont les termes hors diagonale qui font la
différence.
\end{remarque}

\begin{remarque}[Un même $\rho$, plusieurs préparations]\label{rem:ensembles}
La matrice de densité ne garde pas la façon dont le mélange a été préparé. Comme
$\ket{+}\bra{+}+\ket{-}\bra{-}=\mathrm{Id}$ (relation de fermeture, vraie dans toute base orthonormée),
\[
  \tfrac12\ket{0}\bra{0}+\tfrac12\ket{1}\bra{1}
  =\tfrac12\ket{+}\bra{+}+\tfrac12\ket{-}\bra{-}=\frac{\mathrm{Id}}{2}.
\]
Un qubit tiré au hasard parmi $\ket{0},\ket{1}$ et un qubit tiré au hasard parmi $\ket{+},\ket{-}$ ont
la même matrice de densité, donc les mêmes probabilités pour \emph{toute} mesure : aucune expérience
ne distingue ces deux préparations.
\end{remarque}

% ---------------------------------------------------------------------
\subsection{Théorème de décomposition spectrale}\label{sec:spectral}
% ---------------------------------------------------------------------

\begin{resultat}{Décomposition spectrale}
Pour toute matrice de densité $\rho$, il existe une base orthonormée $\{\ket{i}\}$ telle que
\[
  \rho=\sum_i\lambda_i\ket{i}\bra{i},
  \qquad\lambda_i\geq0,\qquad\sum_i\lambda_i=1 .
\]
\end{resultat}

C'est la diagonalisation d'une matrice hermitienne : les $\lambda_i$ sont réelles, elles
sont positives car $\rho$ l'est, et de somme $1$ car $\operatorname{Tr}\rho=1$. Les
$\ket{i}$ sont les vecteurs propres ; les $\lambda_i$ se lisent donc comme des
probabilités : tout état est un mélange d'états purs \emph{orthogonaux}. Un état pur n'a
qu'un seul terme.

\paragraph{Exemple.} Soit
\[
  \rho=\begin{pmatrix}\dfrac12 & \dfrac14\\[0.6ex] \dfrac14 & \dfrac12\end{pmatrix}.
\]
Ses valeurs propres sont $\frac12+\frac14=\frac34$ et $\frac12-\frac14=\frac14$, de
vecteurs propres $\ket{\pm}=\frac{1}{\sqrt2}(\ket{0}\pm\ket{1})$, donc
\[
  \rho=\frac34\ket{+}\bra{+}+\frac14\ket{-}\bra{-},
  \qquad \operatorname{Tr}\rho^2=\frac9{16}+\frac1{16}=\frac58<1 :
\]
$\rho$ est mixte.
Si une valeur propre est dégénérée (comme pour $\mathrm{Id}/2$), la base $\{\ket{i}\}$
n'est pas unique.

\subsection{Sphère de Bloch}\label{sec:bloch}
% ---------------------------------------------------------------------

Un état pur $\alpha\ket{0}+\beta\ket{1}$ contient deux complexes, soit quatre réels. La
normalisation en retire un, la phase globale (inobservable) un autre : il reste
\emph{deux} paramètres réels, ce qui définit un point sur une sphère.

\begin{definition}{Paramétrisation de Bloch}
\[
  \ket{\Psi}=\cos\frac{\theta}{2}\,\ket{0}+\ee^{\ii\varphi}\sin\frac{\theta}{2}\,\ket{1},
  \qquad \theta\in[0,\pi],\quad\varphi\in[0,2\pi[ .
\]
La normalisation est automatique : $\cos^2\frac\theta2+|\ee^{\ii\varphi}|^2\sin^2\frac\theta2=1$
car $|\ee^{\ii\varphi}|=1$. L'état est représenté par le point de la sphère unité de
coordonnées
\[
  \vec r=(\sin\theta\cos\varphi,\ \sin\theta\sin\varphi,\ \cos\theta).
\]
\end{definition}

Aux pôles ($\theta=0$ ou $\pi$) l'angle $\varphi$ n'a pas de sens : il ne reste qu'un
facteur global.

\begin{resultat}{Lecture de la sphère}
\begin{itemize}[nosep]
  \item \textbf{Mesure en base $Z$} : $P(0)=\cos^2\frac\theta2=\frac{1+r_z}{2}$ et
        $P(1)=\sin^2\frac\theta2=\frac{1-r_z}{2}$. L'angle $\varphi$ n'y intervient pas
        (c'est la phase relative).
  \item \textbf{États orthogonaux} $\Leftrightarrow$ \textbf{points antipodaux} :
        $\ket{0}\perp\ket{1}$, $\ket{+}\perp\ket{-}$, $\ket{{+}\ii}\perp\ket{{-}\ii}$.
        Une base orthonormée est donc un diamètre de la sphère.
  \item \textbf{Composantes} : $r_x=\langle X\rangle$, $r_y=\langle Y\rangle$,
        $r_z=\langle Z\rangle$ (moyennes des matrices de Pauli de la
        section~\ref{sec:op-dirac}).
\end{itemize}
\end{resultat}

\paragraph{Boule de Bloch.} Avec $\vec r=(r_x,r_y,r_z)$, la matrice de densité s'écrit
\[
  \rho=\frac12\begin{pmatrix}1+r_z & r_x-\ii r_y\\ r_x+\ii r_y & 1-r_z\end{pmatrix},
  \qquad
  \operatorname{Tr}\rho^2=\frac{1+|\vec r|^2}{2}.
\]
Un état pur ($\operatorname{Tr}\rho^2=1$) vérifie $|\vec r|=1$ : il est \emph{sur} la
sphère. Un état mixte a $|\vec r|<1$ : il est \emph{à l'intérieur} de la boule, et
$\mathrm{Id}/2$ en est le centre ($\vec r=\vec0$, $\operatorname{Tr}\rho^2=\frac12$). Les
exemples de la section~\ref{sec:spectral} donnent $\vec r=(\frac12,0,0)$ pour
$\frac34\ket{+}\bra{+}+\frac14\ket{-}\bra{-}$ et $\vec r=(0,0,\frac12)$ pour
$\mathrm{diag}(\frac34,\frac14)$.

\subsection{Produit tensoriel}\label{sec:tenseur}

Chaque qubit vit dans $\mathcal{H}^2=\mathbb{C}^2$. L'espace du système $(A,B)$ est le
\emph{produit tensoriel}
\[
  \mathcal{H}^2\otimes\mathcal{H}^2=\mathcal{H}^4=\mathbb{C}^4 ,
\]
de dimension $2\times2=4$. Pour $n$ qubits la dimension est $2^n$ :

\midrule
Dimension $2^n$ & $2$ & $4$ & $8$ & $1024$ & $\approx1{,}1\times10^{15}$ \\
\bottomrule
\end{tabular}

\begin{resultat}{Postulat 3 : systèmes composites}
L'espace d'état d'un système composé de deux systèmes d'espaces $\mathcal{H}_A$ et
$\mathcal{H}_B$ est le \gls{tenseur} $\mathcal{H}_A\otimes\mathcal{H}_B$, de dimension
$\dim\mathcal{H}_A\times\dim\mathcal{H}_B$. Si $A$ est dans l'état $\ket{u}$ et $B$ dans l'état
$\ket{v}$, l'ensemble est dans l'état $\ket{u}\otimes\ket{v}$.
\end{resultat}

\begin{definition}{Produit tensoriel de deux vecteurs}
\[
  \begin{pmatrix}a_1\\ a_2\end{pmatrix}\otimes\begin{pmatrix}b_1\\ b_2\end{pmatrix}
  =\begin{pmatrix}a_1\begin{pmatrix}b_1\\ b_2\end{pmatrix}\\[2ex]
                  a_2\begin{pmatrix}b_1\\ b_2\end{pmatrix}\end{pmatrix}
  =\begin{pmatrix}a_1b_1\\ a_1b_2\\ a_2b_1\\ a_2b_2\end{pmatrix}.
\]
Chaque composante du premier vecteur multiplie le second vecteur en entier. On note
$\ket{ij}_{AB}=\ket{i}_A\otimes\ket{j}_B$ : le premier chiffre est l'état de $A$, le second
celui de $B$.
\end{definition}

\subsubsection{Les quatre états de base}

On développe chacun des quatre produits avec $\ket{0}=\begin{pmatrix}1\\ 0\end{pmatrix}$
et $\ket{1}=\begin{pmatrix}0\\ 1\end{pmatrix}$ :
\begin{align*}
  \ket{0}\otimes\ket{0}&=\begin{pmatrix}1\\ 0\end{pmatrix}\otimes\begin{pmatrix}1\\ 0\end{pmatrix}
    =\begin{pmatrix}1\cdot\begin{pmatrix}1\\ 0\end{pmatrix}\\[1.5ex]0\cdot\begin{pmatrix}1\\ 0\end{pmatrix}\end{pmatrix}
    =\begin{pmatrix}1\\ 0\\ 0\\ 0\end{pmatrix}=\ket{00},\\[1.5ex]
  \ket{0}\otimes\ket{1}&=\begin{pmatrix}1\\ 0\end{pmatrix}\otimes\begin{pmatrix}0\\ 1\end{pmatrix}
    =\begin{pmatrix}1\cdot\begin{pmatrix}0\\ 1\end{pmatrix}\\[1.5ex]0\cdot\begin{pmatrix}0\\ 1\end{pmatrix}\end{pmatrix}
    =\begin{pmatrix}0\\ 1\\ 0\\ 0\end{pmatrix}=\ket{01},\\[1.5ex]
  \ket{1}\otimes\ket{0}&=\begin{pmatrix}0\\ 1\end{pmatrix}\otimes\begin{pmatrix}1\\ 0\end{pmatrix}
    =\begin{pmatrix}0\cdot\begin{pmatrix}1\\ 0\end{pmatrix}\\[1.5ex]1\cdot\begin{pmatrix}1\\ 0\end{pmatrix}\end{pmatrix}
    =\begin{pmatrix}0\\ 0\\ 1\\ 0\end{pmatrix}=\ket{10},\\[1.5ex]
  \ket{1}\otimes\ket{1}&=\begin{pmatrix}0\\ 1\end{pmatrix}\otimes\begin{pmatrix}0\\ 1\end{pmatrix}
    =\begin{pmatrix}0\cdot\begin{pmatrix}0\\ 1\end{pmatrix}\\[1.5ex]1\cdot\begin{pmatrix}0\\ 1\end{pmatrix}\end{pmatrix}
    =\begin{pmatrix}0\\ 0\\ 0\\ 1\end{pmatrix}=\ket{11}.
\end{align*}
Le vecteur $\ket{ij}$ a son $1$ en position $2i+j$ (en comptant à partir de $0$) : c'est
la lecture de $ij$ en binaire. Les quatre vecteurs forment la base de calcul de
$\mathcal{H}^4$.

\subsubsection{Produit scalaire et orthonormalité}

\begin{resultat}{Produit scalaire d'un produit tensoriel}
\[
  \big(\bra{a}\otimes\bra{b}\big)\big(\ket{c}\otimes\ket{d}\big)=\braket{a}{c}\,\braket{b}{d}.
\]
\end{resultat}

\emph{Démonstration.} Les composantes de $\ket{a}\otimes\ket{b}$ sont $a_ib_j$, celles de
$\ket{c}\otimes\ket{d}$ sont $c_id_j$. Le produit scalaire est
$\sum_{i,j}(a_ib_j)^*c_id_j=\big(\sum_ia_i^*c_i\big)\big(\sum_jb_j^*d_j\big)=\braket{a}{c}\braket{b}{d}$.
\hfill$\square$

On en déduit $\braket{ij}{kl}=\braket{i}{k}\braket{j}{l}$. Les seize produits scalaires
des états de base sont :
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{@{}c|cccc@{}}
 & $\ket{00}$ & $\ket{01}$ & $\ket{10}$ & $\ket{11}$ \\
\hline
$\bra{00}$ & $1\cdot1=1$ & $1\cdot0=0$ & $0\cdot1=0$ & $0\cdot0=0$ \\
$\bra{01}$ & $1\cdot0=0$ & $1\cdot1=1$ & $0\cdot0=0$ & $0\cdot1=0$ \\
$\bra{10}$ & $0\cdot1=0$ & $0\cdot0=0$ & $1\cdot1=1$ & $1\cdot0=0$ \\
$\bra{11}$ & $0\cdot0=0$ & $0\cdot1=0$ & $1\cdot0=0$ & $1\cdot1=1$ \\
\end{tabular}
\end{center}
(chaque case est $\braket{i}{k}\cdot\braket{j}{l}$). On lit $\braket{ij}{kl}=\delta_{ik}\delta_{jl}$ :
la base est \emph{orthonormée}.

\subsubsection{Produit tensoriel d'états quelconques}

Le produit tensoriel est linéaire en chaque facteur :
$(\lambda\ket{a}+\mu\ket{a'})\otimes\ket{b}=\lambda\ket{a}\otimes\ket{b}+\mu\ket{a'}\otimes\ket{b}$,
et de même pour le second facteur. Pour $\alpha\ket{0}+\beta\ket{1}$ et
$\gamma\ket{0}+\delta\ket{1}$ :
\begin{align*}
  \big(\alpha\ket{0}+\beta\ket{1}\big)\otimes\big(\gamma\ket{0}+\delta\ket{1}\big)
  &=\alpha\ket{0}\otimes\big(\gamma\ket{0}+\delta\ket{1}\big)
    +\beta\ket{1}\otimes\big(\gamma\ket{0}+\delta\ket{1}\big)\\
  &=\alpha\gamma\ket{00}+\alpha\delta\ket{01}+\beta\gamma\ket{10}+\beta\delta\ket{11}.
\end{align*}
Par les composantes, $\begin{pmatrix}\alpha\\ \beta\end{pmatrix}\otimes\begin{pmatrix}\gamma\\ \delta\end{pmatrix}
=\begin{pmatrix}\alpha\gamma\\ \alpha\delta\\ \beta\gamma\\ \beta\delta\end{pmatrix}$ : c'est le même
résultat. La norme est conservée :
\[
  |\alpha\gamma|^2+|\alpha\delta|^2+|\beta\gamma|^2+|\beta\delta|^2
  =\big(|\alpha|^2+|\beta|^2\big)\big(|\gamma|^2+|\delta|^2\big)=1\cdot1=1 .
\]

\paragraph{Exemples.}
\begin{align*}
  \ket{+}\otimes\ket{+}&=\tfrac12\big(\ket{0}+\ket{1}\big)\otimes\big(\ket{0}+\ket{1}\big)
    =\tfrac12\big(\ket{00}+\ket{01}+\ket{10}+\ket{11}\big)
    =\tfrac12\begin{pmatrix}1\\ 1\\ 1\\ 1\end{pmatrix},\\[1ex]
  \ket{+}\otimes\ket{-}&=\tfrac12\big(\ket{0}+\ket{1}\big)\otimes\big(\ket{0}-\ket{1}\big)
    =\tfrac12\big(\ket{00}-\ket{01}+\ket{10}-\ket{11}\big)
    =\tfrac12\begin{pmatrix}1\\ -1\\ 1\\ -1\end{pmatrix},\\[1ex]
  \ket{0}\otimes\ket{+}&=\tfrac{1}{\sqrt2}\big(\ket{00}+\ket{01}\big)
    =\tfrac{1}{\sqrt2}\begin{pmatrix}1\\ 1\\ 0\\ 0\end{pmatrix},\qquad
  \ket{+}\otimes\ket{0}=\tfrac{1}{\sqrt2}\big(\ket{00}+\ket{10}\big)
    =\tfrac{1}{\sqrt2}\begin{pmatrix}1\\ 0\\ 1\\ 0\end{pmatrix}.
\end{align*}
Les deux derniers vecteurs sont différents : \emph{l'ordre des facteurs compte}, le premier
est toujours celui de $A$.

\begin{remarque}[Tous les états ne sont pas des produits]
Un produit $\ket{u}\otimes\ket{v}$ a pour composantes $(\alpha\gamma,\alpha\delta,\beta\gamma,\beta\delta)$,
qui vérifient $c_{00}c_{11}=c_{01}c_{10}$ (les deux membres valent $\alpha\beta\gamma\delta$). Le vecteur
$\frac{1}{\sqrt2}(\ket{00}+\ket{11})$ ne vérifie pas cette relation ($\frac12\neq0$) : ce n'est pas un
produit, mais un état \emph{intriqué} (chapitre~\ref{chap:bell}).
\end{remarque}

\subsubsection{Produit tensoriel de matrices}

\begin{definition}{Produit tensoriel de deux matrices}
\[
  A\otimes B=\begin{pmatrix}A_{11} & A_{12}\\ A_{21} & A_{22}\end{pmatrix}\otimes B
  =\begin{pmatrix}A_{11}B & A_{12}B\\ A_{21}B & A_{22}B\end{pmatrix}
  \qquad(\text{matrice }4\times4).
\]
\end{definition}

\begin{resultat}{Action sur un produit}
\[
  (A\otimes B)\big(\ket{u}\otimes\ket{v}\big)=A\ket{u}\otimes B\ket{v}.
\]
\end{resultat}

\emph{Démonstration.} Avec $\ket{u}=(u_1,u_2)^{\mathsf T}$, le vecteur $\ket{u}\otimes\ket{v}$ a
pour blocs $u_k\ket{v}$ ($k=1,2$). Le bloc de lignes $i$ de $(A\otimes B)(\ket{u}\otimes\ket{v})$ est
donc
\[
  \sum_kA_{ik}B\,\big(u_k\ket{v}\big)=\Big(\sum_kA_{ik}u_k\Big)B\ket{v}=(A\ket{u})_i\,B\ket{v},
\]
qui est le bloc $i$ de $A\ket{u}\otimes B\ket{v}$. \hfill$\square$

\begin{aretenir}
\begin{itemize}
  \item Un opérateur est une matrice. Il est \emph{hermitien} ($A^\dagger=A$) s'il représente une
        grandeur mesurable : ses valeurs propres sont réelles, et $\langle A\rangle=\bra{\Psi}A\ket{\Psi}$.
        Il est \emph{unitaire} ($U^\dagger U=\mathrm{Id}$) s'il conserve la norme : c'est une porte.
  \item La matrice de densité $\rho=\ket{\Psi}\bra{\Psi}$ est hermitienne, positive et de trace $1$.
        Sa diagonale donne les probabilités (populations), ses termes hors diagonale la phase relative
        (cohérences). Valeur moyenne : $\langle A\rangle=\operatorname{Tr}(\rho A)$.
  \item Un état est \emph{pur} si $\operatorname{Tr}\rho^2=1$, \emph{mixte} si $\operatorname{Tr}\rho^2<1$.
        Un mélange statistique n'est pas une superposition : $\mathrm{Id}/2\neq\ket{+}\bra{+}$.
  \item Sur la sphère de Bloch, un état pur est un point de coordonnées
        $\vec r=(\langle X\rangle,\langle Y\rangle,\langle Z\rangle)$ ; deux états orthogonaux sont
        antipodaux ; un état mixte est dans la boule, et $\mathrm{Id}/2$ en est le centre.
  \item $n$ qubits vivent dans $\mathcal{H}^{\otimes n}$, de dimension $2^n$. Le produit
        $\ket{u}\otimes\ket{v}$ multiplie les amplitudes, et
        $(A\otimes B)(\ket{u}\otimes\ket{v})=A\ket{u}\otimes B\ket{v}$. Tous les états de
        $\mathcal{H}^4$ ne sont pas des produits.
\end{itemize}
\end{aretenir}
